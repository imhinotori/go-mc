    func (t *TickLoop) Run(ctx context.Context, inbound <-chan Intent) {
        const wake = 5 * time.Millisecond
        const step = 50 * time.Millisecond
        const maxFrame = 250 * time.Millisecond // spiral clamp
        ticker := time.NewTicker(wake); defer ticker.Stop()
        last := t.clock.Now(); var acc time.Duration
        for {
            select {
            case <-ctx.Done(): return
            case <-ticker.C:
                now := t.clock.Now()
                frame := now.Sub(last); last = now
                if frame > maxFrame { frame = maxFrame } // drop backlog; slow, never freeze
                acc += frame
                for acc >= step {
                    t.drainInbound(inbound) // non-blocking drain on the OWNER goroutine
                    t.tickOnce()            // ordered phases + gametime++ + recordMSPT
                    acc -= step
                }
            }
        }
    }
