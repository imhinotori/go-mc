```go
func (pf *PlacedFeature) place(ctx *PlacementContext, rng RandomSource, origin BlockPos) bool {
    positions := []BlockPos{origin}
    for _, mod := range pf.placement {            // ordered; ONE rng threaded through
        var next []BlockPos
        for _, p := range positions {
            next = append(next, mod.getPositions(ctx, rng, p)...)  // flatMap, sequential
        }
        positions = next
    }
    placed := false
    for _, p := range positions {
        if pf.configured.place(ctx, rng, p) { placed = true }   // configured-feature body (11-03/12 wires the real call)
    }
    return placed
}
```
